cmake_minimum_required(VERSION 3.16)

# ton-tvm-executor serves ton.tvm.v1.Executor over gRPC, which needs grpc++
# and grpc_cpp_plugin (libgrpc++-dev, protobuf-compiler-grpc): where they are
# missing, it is skipped.
find_package(PkgConfig)
if (PkgConfig_FOUND)
  pkg_check_modules(GRPCPP IMPORTED_TARGET grpc++)
endif()
find_program(GRPC_CPP_PLUGIN grpc_cpp_plugin)
if (NOT GRPCPP_FOUND OR NOT GRPC_CPP_PLUGIN)
  message(STATUS "ton-tvm-executor is skipped: grpc++ or grpc_cpp_plugin is not found")
  return()
endif()
find_package(Protobuf REQUIRED)

# sf/ton/type/v1/type.proto, which these import, is generated by
# ton-index-firehose into ton-firehose-proto, so that its code is linked once.
set(EXECUTOR_PROTO_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../ton-index-firehose/proto)
set(EXECUTOR_PROTO_OUT ${CMAKE_CURRENT_BINARY_DIR}/proto)
set(EXECUTOR_PROTO_SRCS)
foreach(proto ton/state/v1/state.proto ton/tvm/v1/tvm.proto ton/tvm/v1/executor.proto)
  string(REGEX REPLACE "\\.proto$" "" proto_base ${proto})
  list(APPEND EXECUTOR_PROTO_SRCS ${EXECUTOR_PROTO_OUT}/${proto_base}.pb.cc)
  add_custom_command(
    OUTPUT ${EXECUTOR_PROTO_OUT}/${proto_base}.pb.cc ${EXECUTOR_PROTO_OUT}/${proto_base}.pb.h
    COMMAND ${CMAKE_COMMAND} -E make_directory ${EXECUTOR_PROTO_OUT}
    COMMAND ${Protobuf_PROTOC_EXECUTABLE} --experimental_allow_proto3_optional
            --proto_path=${EXECUTOR_PROTO_DIR} --proto_path=${Protobuf_INCLUDE_DIRS}
            --cpp_out=${EXECUTOR_PROTO_OUT} ${proto}
    DEPENDS ${EXECUTOR_PROTO_DIR}/${proto}
    COMMENT "Generating C++ for ${proto}")
endforeach()
add_custom_command(
  OUTPUT ${EXECUTOR_PROTO_OUT}/ton/tvm/v1/executor.grpc.pb.cc ${EXECUTOR_PROTO_OUT}/ton/tvm/v1/executor.grpc.pb.h
  COMMAND ${CMAKE_COMMAND} -E make_directory ${EXECUTOR_PROTO_OUT}
  COMMAND ${Protobuf_PROTOC_EXECUTABLE} --proto_path=${EXECUTOR_PROTO_DIR} --proto_path=${Protobuf_INCLUDE_DIRS}
          --grpc_out=${EXECUTOR_PROTO_OUT} --plugin=protoc-gen-grpc=${GRPC_CPP_PLUGIN} ton/tvm/v1/executor.proto
  DEPENDS ${EXECUTOR_PROTO_DIR}/ton/tvm/v1/executor.proto
  COMMENT "Generating gRPC C++ for ton/tvm/v1/executor.proto")

# The messages, used on both sides: protobuf 3 does not use Abseil.
add_library(ton-tvm-executor-proto STATIC ${EXECUTOR_PROTO_SRCS})
target_include_directories(ton-tvm-executor-proto PUBLIC ${EXECUTOR_PROTO_OUT})
target_link_libraries(ton-tvm-executor-proto PUBLIC ton-firehose-proto)

# The gRPC server, apart from TON: grpc++ is built with the system Abseil and
# TON with its own, so no source sees the headers of both. grpc++ is linked
# privately, so that its include directories do not reach TON sources.
add_library(ton-tvm-executor-grpc STATIC
    src/ExecutorServer.cpp
    ${EXECUTOR_PROTO_OUT}/ton/tvm/v1/executor.grpc.pb.cc)
target_compile_features(ton-tvm-executor-grpc PRIVATE cxx_std_17)
target_link_libraries(ton-tvm-executor-grpc PUBLIC ton-tvm-executor-proto PRIVATE PkgConfig::GRPCPP)

# TON builds its own static OpenSSL; system gRPC uses the system shared one.
# Pull the complete static libraries and hide their symbols from shared libs.
# Otherwise EVP objects can cross versions (e.g. CHKSIG crashes in verify).
function(executor_crypto_link target)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND OPENSSL_CRYPTO_LIBRARY MATCHES "\\.a$"
      AND OPENSSL_SSL_LIBRARY MATCHES "\\.a$")
    target_link_libraries(${target} PRIVATE -Wl,--whole-archive
        ${OPENSSL_SSL_LIBRARY} ${OPENSSL_CRYPTO_LIBRARY} -Wl,--no-whole-archive)
    target_link_options(${target} PRIVATE -Wl,--exclude-libs,libssl.a:libcrypto.a)
  endif()
endfunction()

set(EXECUTOR_CALL_SRCS src/Calls.cpp src/GetMethod.cpp src/Emulate.cpp src/Replay.cpp src/Stack.cpp)
add_executable(ton-tvm-executor src/main.cpp src/Executor.cpp ${EXECUTOR_CALL_SRCS})
target_compile_features(ton-tvm-executor PRIVATE cxx_std_20)
target_link_libraries(ton-tvm-executor PRIVATE ton-tvm-executor-grpc ton-firehose-convert emulator_static)
target_link_options(ton-tvm-executor PUBLIC -rdynamic)
executor_crypto_link(ton-tvm-executor)

add_executable(test-ton-tvm-calls test/calls.cpp ${EXECUTOR_CALL_SRCS})
target_compile_features(test-ton-tvm-calls PRIVATE cxx_std_20)
target_include_directories(test-ton-tvm-calls PRIVATE src)
target_link_libraries(test-ton-tvm-calls PRIVATE ton-tvm-executor-proto ton-firehose-convert emulator_static)
add_test(NAME test-ton-tvm-calls COMMAND test-ton-tvm-calls)

add_executable(test-ton-tvm-link test/link.cpp)
target_include_directories(test-ton-tvm-link PRIVATE src)
target_link_libraries(test-ton-tvm-link PRIVATE ton-tvm-executor-grpc ton_crypto_core)
executor_crypto_link(test-ton-tvm-link)
add_test(NAME test-ton-tvm-link COMMAND test-ton-tvm-link)

install(TARGETS ton-tvm-executor RUNTIME DESTINATION bin)
